1. Итоговый балл, порог и решение по каждой заявке.

PREFIX credit: <http://www.semanticweb.org/creditworthiness#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?profile ?score ?threshold ?decisionLabel
WHERE {
  ?a a credit:CreditworthinessAssessment ; credit:hasPersonalData ?profile ;
     credit:approvalThreshold ?threshold ; credit:profilePoints ?profilePoints ;
     credit:affordabilityPoints ?affordability ; credit:incomePoints ?income ;
     credit:historyPoints ?history ; credit:savingsPoints ?savings ;
     credit:debtPoints ?debt ; credit:riskAdjustment ?adjustment .
  BIND (?profilePoints + ?affordability + ?income + ?history + ?savings + ?debt + ?adjustment AS ?score)
  BIND (IF(?score >= ?threshold, credit:GrantCredit, credit:DoNotGrantCredit) AS ?decision)
  ?decision rdfs:label ?decisionLabel .
  FILTER (LANG(?decisionLabel) = "ru")
}
ORDER BY DESC(?score) ?profile
